**Utilizando a biblioteca `redis` para cache**

import redis
import json
from datetime import datetime, timedelta

# Configurações do cache
CACHE_HOST = 'localhost'
CACHE_PORT = 6379
CACHE_DB = 0

# Cria uma conexão com o Redis
redis_client = redis.Redis(host=CACHE_HOST, port=CACHE_PORT, db=CACHE_DB)

# Função para obter o TTL (Time To Live) para cada tipo de consulta
def get_ttl(consulta_tipo):
    ttl_map = {
        'produto': 60 * 60 * 24,  # 1 dia
        'categoria': 60 * 60 * 24 * 7,  # 1 semana
        'busca': 60 * 60,  # 1 hora
        'recomendacao': 60 * 30,  # 30 minutos
    }
    return ttl_map.get(consulta_tipo, 60 * 60)  # padrão: 1 hora

# Função para obter a resposta de uma consulta do LLM
def obter_resposta_llm(consulta, tipo):
    # Verifica se a resposta está no cache
    cache_key = f"llm:{tipo}:{consulta}"
    resposta = redis_client.get(cache_key)
    if resposta:
        return json.loads(resposta)

    ttl = get_ttl(tipo)
    redis_client.setex(cache_key, ttl, json.dumps(resposta_llm))
